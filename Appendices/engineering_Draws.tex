
% Create numbered section but hide header and spacing
\makeatletter
\newcommand{\invisiblesection}[1]{%
	\refstepcounter{section}% increase section counter
	\addcontentsline{toc}{section}{#1}% optional: add to TOC
	\phantomsection % for hyperref anchors
	% No header text printed, no vertical space
	\def\@currentlabel{\thesection}% for \ref to work correctly
}
\makeatother
